⚠️ Script SQL para o Supabase

Acesse o SQL Editor do seu painel Supabase e execute este comando para criar a tabela de sugestões:

create table if not exists public.sugestao_ponto (
  id bigint generated by default as identity primary key,
  "linhaId" integer not null,
  toque text,
  titulo text,
  letra text,
  autor text,
  link text,
  autoral boolean default false,
  created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

alter table public.sugestao_ponto enable row level security;
create policy "Acesso total publico sugestoes" on public.sugestao_ponto for all using (true) with check (true);
